# Report Q&A Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an inline, report-scoped multi-turn Q&A entrypoint to the Financial Reports page, with bounded session memory, evidence-backed answers, and automatic session reset when the active report changes.

**Architecture:** The frontend owns the visible transcript, recent turns, and compressed session summary. The backend owns only ephemeral report context, keyed by `report_key`, so it can answer follow-up questions without re-downloading the report for every turn. Answers prefer `llm_hybrid` mode when a model is available and fall back to a constrained rule-based response built from current report metrics, `Three Questions`, and `LLM Reading Notes`.

**Tech Stack:** FastAPI, Pydantic, plain HTML/JS, in-process Python memory cache, DeepSeek-compatible OpenAI chat completions, pytest.

## Implementation Notes Locked Before Coding

- Q&A stays disabled after plain `Load Financial Report` because that path does not load report text. It becomes enabled only after:
  - `Auto Read Annual Report`, or
  - `Analyze Report URL`
- Report context is cached only in backend process memory, never in SQLite.
- Chat transcript is not stored server-side; the frontend sends:
  - recent `history`
  - `session_summary`
  - `question`
  - `report_key`
- The backend rejects any Q&A request whose `report_key` is not known or whose `symbol` does not match the cached context.
- `Analyze Report URL` will start sending `symbol` together with `url` so URL-based report sessions can share the same `report_key` contract as auto-read sessions.

```python
from __future__ import annotations

import threading
from typing import Any


_REPORT_CONTEXT_LOCK = threading.Lock()
_REPORT_CONTEXTS: dict[str, dict[str, Any]] = {}


def build_report_key(symbol: str, report: dict[str, Any] | None) -> str | None:
    """Return a stable key for one active report session."""
    if not symbol or not report:
        return None
    source_url = report.get("document_url") or report.get("detail_url")
    if not source_url:
        return None
    return f"{symbol}|{source_url}"


def clear_report_context_cache() -> None:
    """Clear in-memory report contexts for tests and process resets."""
    with _REPORT_CONTEXT_LOCK:
        _REPORT_CONTEXTS.clear()


def store_report_context(report_key: str, context: dict[str, Any]) -> None:
    """Store one active report context in process memory."""
    with _REPORT_CONTEXT_LOCK:
        _REPORT_CONTEXTS[report_key] = dict(context)


def get_cached_report_context(report_key: str) -> dict[str, Any] | None:
    """Return one cached report context if present."""
    with _REPORT_CONTEXT_LOCK:
        current = _REPORT_CONTEXTS.get(report_key)
    return dict(current) if current else None
```

```python
def bound_history(
    history: list[dict[str, str]],
    session_summary: str,
    *,
    max_turns: int = 6,
) -> tuple[list[dict[str, str]], str]:
    """Keep recent turns raw and compress older turns into the carried summary."""
    normalized = [
        {"role": item["role"], "content": str(item["content"]).strip()}
        for item in history
        if item.get("role") in {"user", "assistant"} and str(item.get("content") or "").strip()
    ]
    if len(normalized) <= max_turns:
        return normalized, session_summary.strip()

    older = normalized[:-max_turns]
    recent = normalized[-max_turns:]
    compressed = " | ".join(f'{item["role"]}: {item["content"][:120]}' for item in older)
    merged = " ".join(part for part in [session_summary.strip(), compressed] if part).strip()
    return recent, merged


def build_rule_fallback_answer(question: str, context: dict[str, Any]) -> dict[str, Any]:
    """Return a constrained answer using only current extracted metrics and prior assessment output."""
    answers = context.get("answers") or []
    llm_analysis = context.get("llm_analysis") or {}
    matching = next((item for item in answers if any(token in question for token in item.get("question", ""))), None)
    evidence = list((matching or {}).get("evidence") or [])
    if not evidence and llm_analysis.get("summary"):
        evidence.append(llm_analysis["summary"])

    short_answer = (matching or {}).get("summary") or "当前报告已加载，但现阶段只能基于已抽取指标和现有评估结果回答该问题。"
    return {
        "mode": "rule_fallback",
        "short_answer": short_answer,
        "evidence": evidence[:5],
        "citations": [],
        "confidence": "medium" if evidence else "low",
    }
```

```python
from app.services.llm_service import answer_report_question_with_llm


def answer_report_question(
    *,
    symbol: str,
    report_key: str,
    question: str,
    history: list[dict[str, str]],
    session_summary: str,
    use_llm: bool,
) -> dict[str, Any]:
    """Return one report-scoped answer using cached report context plus bounded chat history."""
    context = get_cached_report_context(report_key)
    if not context:
        raise ValueError("Active report context not found. Reload the report and try again.")
    if context.get("symbol") != symbol:
        raise ValueError("Report key does not match the requested symbol.")

    bounded_history, updated_summary = bound_history(history, session_summary)
    if use_llm:
        try:
            llm_answer = answer_report_question_with_llm(
                report_title=context.get("report", {}).get("title"),
                report_text=context.get("report_text", ""),
                question=question,
                history=bounded_history,
                session_summary=updated_summary,
                extracted_metrics=context.get("extracted_metrics", {}),
                answers=context.get("answers", []),
                llm_analysis=context.get("llm_analysis"),
            )
            llm_answer["mode"] = "llm_hybrid"
            llm_answer["updated_session_summary"] = updated_summary
            llm_answer["session_reset"] = False
            llm_answer["session_key"] = report_key
            return llm_answer
        except Exception:
            pass

    fallback = build_rule_fallback_answer(question, context)
    fallback["updated_session_summary"] = updated_summary
    fallback["session_reset"] = False
    fallback["session_key"] = report_key
    return fallback
```

```python
from pydantic import BaseModel, Field


class FinancialReportQaTurn(BaseModel):
    role: Literal["user", "assistant"]

    url: str
    symbol: str | None = None


class FinancialReportQaRequest(BaseModel):
    symbol: str
    report_key: str
    question: str = Field(..., min_length=1, max_length=1000)
    history: list[FinancialReportQaTurn] = []
    session_summary: str = ""
    use_llm: bool = True


@router.post("/api/financial-report-qa")
def financial_report_qa(payload: FinancialReportQaRequest) -> dict:
```

```javascript
async function askActiveReport() {
  const question = reportQaInputEl.value.trim();
  if (!question || !reportQaSession.reportKey || !reportQaSession.symbol) return;

  appendReportQaTurn("user", question, null);
  reportQaAskBtn.disabled = true;

  try {
    const res = await fetch(`${apiBase}/api/financial-report-qa`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        symbol: reportQaSession.symbol,
        report_key: reportQaSession.reportKey,
        question,
        history: reportQaSession.history,
        session_summary: reportQaSession.sessionSummary,
        use_llm: true,
      }),
    });
    const data = await parseJsonResponse(res, "Financial report Q&A endpoint");
    if (!res.ok) throw new Error(data.detail || "Request failed");

    reportQaSession.history.push({ role: "user", content: question });
    reportQaSession.history.push({ role: "assistant", content: data.short_answer || "-" });
    reportQaSession.history = reportQaSession.history.slice(-6);
    reportQaSession.sessionSummary = data.updated_session_summary || reportQaSession.sessionSummary;
    appendReportQaTurn("assistant", data.short_answer || "-", data);
    reportQaInputEl.value = "";
  } finally {
    reportQaAskBtn.disabled = false;
  }
}
```
